cmake_minimum_required(VERSION 3.24)
project(cryptopro_native_bridge VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "CryptoPro Native Bridge is supported on Windows only.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure this project for x64. Win32 builds are not supported by this target.")
endif()

set(CRYPTOPRO_SDK_ROOT "C:/Program Files (x86)/Crypto Pro/SDK" CACHE PATH "CryptoPro SDK root")
set(CRYPTOPRO_INCLUDE_DIR "${CRYPTOPRO_SDK_ROOT}/include")
set(CRYPTOPRO_LIBRARY "${CRYPTOPRO_SDK_ROOT}/lib/amd64/cades.lib")

if(NOT EXISTS "${CRYPTOPRO_INCLUDE_DIR}/cades.h")
  message(FATAL_ERROR "cades.h not found: ${CRYPTOPRO_INCLUDE_DIR}. Set CRYPTOPRO_SDK_ROOT.")
endif()
if(NOT EXISTS "${CRYPTOPRO_LIBRARY}")
  message(FATAL_ERROR "x64 cades.lib not found: ${CRYPTOPRO_LIBRARY}")
endif()

add_executable(cryptopro_native_bridge src/main.cpp)
target_compile_features(cryptopro_native_bridge PRIVATE cxx_std_17)
target_include_directories(cryptopro_native_bridge PRIVATE "${CRYPTOPRO_INCLUDE_DIR}")
target_link_libraries(cryptopro_native_bridge PRIVATE "${CRYPTOPRO_LIBRARY}" Crypt32 Advapi32 Ncrypt)
target_compile_definitions(cryptopro_native_bridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(cryptopro_native_bridge PRIVATE /W4 /permissive- /EHsc /utf-8)
  set_property(TARGET cryptopro_native_bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
set_target_properties(cryptopro_native_bridge PROPERTIES OUTPUT_NAME "cryptopro_native_bridge")
